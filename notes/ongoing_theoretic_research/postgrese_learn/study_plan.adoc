= PostgreSQL Practical Study Plan
:toc: left
:toclevels: 3

A hands-on, project-driven study plan for learning PostgreSQL from basics to production-grade usage.
Each phase builds on the previous one. Implement the sample project as you learn each concept.

== Phase 1 — Foundation (Week 1)

=== Goals
- Connect to PostgreSQL and navigate psql
- Understand databases, schemas, tables, and basic data types
- Write basic CRUD queries

=== Concepts
- `psql` CLI basics: `\l`, `\c`, `\dt`, `\d tablename`, `\i file.sql`
- Data types: `TEXT`, `VARCHAR`, `INTEGER`, `BIGINT`, `NUMERIC`, `BOOLEAN`, `DATE`, `TIMESTAMP`, `UUID`
- DDL: `CREATE`, `ALTER`, `DROP`
- DML: `INSERT`, `UPDATE`, `DELETE`, `SELECT`
- Constraints: `NOT NULL`, `UNIQUE`, `PRIMARY KEY`, `FOREIGN KEY`, `CHECK`, `DEFAULT`

=== Sample Project: Library Management System

[source,sql]
----
-- schema
CREATE TABLE authors (
    id          SERIAL PRIMARY KEY,
    name        TEXT NOT NULL,
    birth_year  INTEGER CHECK (birth_year > 1000)
);

CREATE TABLE books (
    id           SERIAL PRIMARY KEY,
    title        TEXT NOT NULL,
    author_id    INTEGER REFERENCES authors(id) ON DELETE CASCADE,
    isbn         VARCHAR(13) UNIQUE,

    id         SERIAL PRIMARY KEY,
    email      TEXT NOT NULL UNIQUE,

CREATE TABLE loans (
    id          SERIAL PRIMARY KEY,
    book_id     INTEGER REFERENCES books(id),
    member_id   INTEGER REFERENCES members(id),
    loaned_at   TIMESTAMP DEFAULT NOW(),
    returned_at TIMESTAMP
);
----

=== Practice Queries (Library System continued)

[source,sql]
----
-- Books currently on loan (not yet returned)
SELECT b.title, m.email, l.loaned_at
FROM loans l
JOIN books b ON b.id = l.book_id
JOIN members m ON m.id = l.member_id
WHERE l.returned_at IS NULL;

-- Members who have never loaned a book
SELECT m.*
FROM members m
LEFT JOIN loans l ON l.member_id = m.id
WHERE l.id IS NULL;

-- Top 3 most loaned books
SELECT b.title, COUNT(l.id) AS loan_count
FROM books b
LEFT JOIN loans l ON l.book_id = b.id
GROUP BY b.id
ORDER BY loan_count DESC
LIMIT 3;

-- Average loan duration in days (for returned books)
SELECT b.title,
       AVG(returned_at - loaned_at) AS avg_duration
FROM loans l
JOIN books b ON b.id = l.book_id
WHERE l.returned_at IS NOT NULL
GROUP BY b.id;

-- Books overdue (loaned > 14 days, not returned)
SELECT b.title, m.email, l.loaned_at,
       NOW() - l.loaned_at AS overdue_by
FROM loans l
JOIN books b ON b.id = l.book_id
JOIN members m ON m.id = l.member_id
WHERE l.returned_at IS NULL
  AND l.loaned_at < NOW() - INTERVAL '14 days';
----

=== New Project: E-Commerce Orders

[source,sql]
----
CREATE TABLE customers (
    id         SERIAL PRIMARY KEY,
    name       TEXT NOT NULL,
    country    TEXT NOT NULL
);

CREATE TABLE products (
    id       SERIAL PRIMARY KEY,
    name     TEXT NOT NULL,
    category TEXT NOT NULL,
    price    NUMERIC(10,2) NOT NULL
);

CREATE TABLE orders (
    id           SERIAL PRIMARY KEY,
    customer_id  INTEGER REFERENCES customers(id),
    ordered_at   TIMESTAMP DEFAULT NOW(),
    status       TEXT CHECK (status IN ('pending','shipped','delivered','cancelled'))
);

CREATE TABLE order_items (
    id         SERIAL PRIMARY KEY,
    order_id   INTEGER REFERENCES orders(id),
    product_id INTEGER REFERENCES products(id),
    quantity   INTEGER NOT NULL,
    unit_price NUMERIC(10,2) NOT NULL
);
----

[source,sql]
----
-- CTE: revenue per customer with their rank
WITH customer_revenue AS (
    SELECT c.id, c.name,
           SUM(oi.quantity * oi.unit_price) AS total_spent
    FROM customers c
    JOIN orders o ON o.customer_id = c.id
    JOIN order_items oi ON oi.order_id = o.id
    WHERE o.status = 'delivered'
    GROUP BY c.id
)
SELECT *,
       RANK() OVER (ORDER BY total_spent DESC) AS revenue_rank
FROM customer_revenue;

-- Window: month-over-month revenue growth
WITH monthly AS (
    SELECT DATE_TRUNC('month', ordered_at) AS month,
           SUM(oi.quantity * oi.unit_price)  AS revenue
    FROM orders o
    JOIN order_items oi ON oi.order_id = o.id
    WHERE o.status != 'cancelled'
    GROUP BY 1
)
SELECT month, revenue,
       LAG(revenue) OVER (ORDER BY month) AS prev_month,
       revenue - LAG(revenue) OVER (ORDER BY month) AS diff
FROM monthly;

-- Recursive CTE: product category hierarchy
CREATE TABLE categories (
    id        SERIAL PRIMARY KEY,
    name      TEXT NOT NULL,
    parent_id INTEGER REFERENCES categories(id)
);

WITH RECURSIVE category_tree AS (
    SELECT id, name, parent_id, 0 AS depth, name::TEXT AS path
    FROM categories WHERE parent_id IS NULL
    UNION ALL
    SELECT c.id, c.name, c.parent_id, ct.depth + 1,
           ct.path || ' > ' || c.name
    FROM categories c
    JOIN category_tree ct ON ct.id = c.parent_id
)
SELECT * FROM category_tree ORDER BY path;
----

== Phase 4 — Indexes and Performance (Week 4)

=== Goals
- Understand how PostgreSQL plans and executes queries
- Create the right indexes for your workload
- Read and interpret `EXPLAIN ANALYZE`

=== Concepts
- `EXPLAIN`, `EXPLAIN ANALYZE`, `EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON)`
- Sequential scan vs index scan vs index-only scan vs bitmap scan
- B-tree indexes (default)
- Partial indexes
- Composite indexes and column order
- `GIN` indexes for full-text search and JSONB
- `BRIN` indexes for time-series / ordered data
- Index bloat and `REINDEX`
- `pg_stat_user_indexes`, `pg_stat_user_tables`

=== Practice

[source,sql]
----
-- Identify slow query
EXPLAIN ANALYZE
SELECT * FROM order_items WHERE unit_price > 100;

-- Create index
CREATE INDEX idx_order_items_unit_price ON order_items(unit_price);

-- Partial index: only index active/pending orders (saves space)
CREATE INDEX idx_orders_pending
    ON orders(customer_id)
    WHERE status = 'pending';

-- Composite index: queries filter by customer then sort by date
CREATE INDEX idx_orders_customer_date
    ON orders(customer_id, ordered_at DESC);

-- Check index usage
SELECT relname, indexrelname, idx_scan, idx_tup_read
FROM pg_stat_user_indexes
WHERE relname = 'orders';

-- Find tables doing too many sequential scans (candidates for indexing)
SELECT relname, seq_scan, idx_scan
FROM pg_stat_user_tables
ORDER BY seq_scan DESC;
----

*Practice tasks:*

- Run `EXPLAIN ANALYZE` before and after adding an index. Document the cost difference.
- Identify a query from Phase 3 that benefits from a composite index.
- Create a partial index and verify it is used in the query plan.

== Phase 5 — Transactions, Locks, and Concurrency (Week 5)

=== Goals
- Understand ACID guarantees in PostgreSQL
- Write safe concurrent code
- Understand isolation levels and what anomalies each prevents

=== Concepts
- `BEGIN`, `COMMIT`, `ROLLBACK`, `SAVEPOINT`
- Isolation levels: Read Committed (default), Repeatable Read, Serializable
- Anomalies: dirty read, non-repeatable read, phantom read
- Row-level locking: `SELECT ... FOR UPDATE`, `FOR SHARE`, `SKIP LOCKED`
- Advisory locks
- Deadlocks: detection, prevention, `deadlock_timeout`
- MVCC (Multi-Version Concurrency Control) — how Postgres avoids reader-writer blocks

=== Practice Scenarios

[source,sql]
----
-- Safe bank transfer with explicit transaction
CREATE TABLE accounts (
    id      SERIAL PRIMARY KEY,
    owner   TEXT NOT NULL,
    balance NUMERIC(12,2) NOT NULL CHECK (balance >= 0)
);

INSERT INTO accounts (owner, balance) VALUES ('Alice', 1000), ('Bob', 500);

BEGIN;
    UPDATE accounts SET balance = balance - 200 WHERE owner = 'Alice';
    UPDATE accounts SET balance = balance + 200 WHERE owner = 'Bob';
COMMIT;

-- SAVEPOINT: partial rollback
BEGIN;
    INSERT INTO orders (customer_id, status) VALUES (1, 'pending');
    SAVEPOINT after_order;

    INSERT INTO order_items (order_id, product_id, quantity, unit_price)
    VALUES (currval('orders_id_seq'), 999, 1, 50.00); -- invalid product_id

ROLLBACK TO after_order;   -- undo bad item, keep order
COMMIT;

-- SKIP LOCKED: job-queue pattern (workers claim tasks without blocking each other)
SELECT * FROM orders
WHERE status = 'pending'
ORDER BY ordered_at
LIMIT 1
FOR UPDATE SKIP LOCKED;
----

*Experiment:*

Open two `psql` sessions. In session 1, start a transaction and lock a row with `FOR UPDATE`. In session 2, try to update the same row and observe the block. Then commit session 1 and watch session 2 proceed.

=== New Project: Product Catalog with Flexible Attributes

[source,sql]
----
CREATE TABLE product_catalog (
    id         SERIAL PRIMARY KEY,
    name       TEXT NOT NULL,
    category   TEXT NOT NULL,
    attributes JSONB NOT NULL DEFAULT '{}'
);

INSERT INTO product_catalog (name, category, attributes) VALUES
('Laptop Pro', 'electronics', '{"brand":"Dell","ram_gb":16,"storage_gb":512,"color":"silver"}'),
('Running Shoes', 'footwear', '{"brand":"Nike","size":42,"color":"black","waterproof":true}'),
('Coffee Maker', 'appliances', '{"brand":"Breville","cups":12,"color":"stainless","programmable":true}');

-- Query by nested attribute
SELECT name FROM product_catalog
WHERE attributes->>'brand' = 'Nike';

-- Containment: find products with color = black
SELECT name FROM product_catalog
WHERE attributes @> '{"color":"black"}';

-- Filter electronics with >= 16GB RAM
SELECT name, attributes->>'ram_gb' AS ram
FROM product_catalog
WHERE category = 'electronics'
  AND (attributes->>'ram_gb')::INTEGER >= 16;

-- GIN index for fast JSONB containment queries
CREATE INDEX idx_product_attributes ON product_catalog USING GIN (attributes);

-- Expand JSONB to rows (useful for analytics)
SELECT name, key, value
FROM product_catalog,
     jsonb_each_text(attributes)
WHERE category = 'electronics';

-- Update a single key without replacing the whole object
UPDATE product_catalog
SET attributes = jsonb_set(attributes, '{ram_gb}', '32')
WHERE name = 'Laptop Pro';
----

== Phase 7 — Full-Text Search (Week 7)

=== Goals
- Implement full-text search without an external search engine
- Understand `tsvector` and `tsquery`
- Rank and highlight results

=== Concepts
- `tsvector`: document representation (normalized lexemes)
- `tsquery`: search query
- `to_tsvector(config, text)`, `to_tsquery()`, `plainto_tsquery()`, `websearch_to_tsquery()`
- `@@` match operator
- `ts_rank()`, `ts_headline()`
- Persisted `tsvector` column with trigger for auto-update
- `GIN` index on `tsvector`

=== Practice: Article Search

[source,sql]
----
CREATE TABLE articles (
    id         SERIAL PRIMARY KEY,
    title      TEXT NOT NULL,
    body       TEXT NOT NULL,
    search_vec TSVECTOR
);

-- Populate search vector
UPDATE articles
SET search_vec = to_tsvector('english', title || ' ' || body);

-- GIN index
CREATE INDEX idx_articles_search ON articles USING GIN (search_vec);

-- Auto-update with trigger
CREATE FUNCTION update_search_vec() RETURNS TRIGGER AS $$
BEGIN
    NEW.search_vec := to_tsvector('english', NEW.title || ' ' || NEW.body);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_articles_search
BEFORE INSERT OR UPDATE ON articles
FOR EACH ROW EXECUTE FUNCTION update_search_vec();

-- Search
SELECT title, ts_rank(search_vec, query) AS rank,
       ts_headline('english', body, query) AS snippet
FROM articles,
     websearch_to_tsquery('english', 'postgresql index performance') query
WHERE search_vec @@ query
ORDER BY rank DESC;
----

== Phase 8 — Stored Procedures, Functions, and Triggers (Week 8)

=== Goals
- Write reusable logic inside the database
- Use triggers to enforce business rules automatically

=== Concepts
- `CREATE FUNCTION` with PL/pgSQL

- `IN`, `OUT`, `INOUT` parameters
- `CREATE PROCEDURE` (PostgreSQL 11+, supports transactions)
- `EXCEPTION` handling in PL/pgSQL
- Triggers: `BEFORE` / `AFTER`, `FOR EACH ROW` / `FOR EACH STATEMENT`
- `NEW`, `OLD` records in triggers

=== Practice

[source,sql]
----
-- Function: get all overdue loans for a member
CREATE FUNCTION get_overdue_loans(p_member_id INTEGER)
RETURNS TABLE (book_title TEXT, days_overdue INTEGER) AS $$
BEGIN
    RETURN QUERY
    SELECT b.title,
           EXTRACT(DAY FROM NOW() - l.loaned_at)::INTEGER
    FROM loans l
    JOIN books b ON b.id = l.book_id
    WHERE l.member_id = p_member_id
      AND l.returned_at IS NULL
      AND l.loaned_at < NOW() - INTERVAL '14 days';
END;
$$ LANGUAGE plpgsql;

SELECT * FROM get_overdue_loans(1);

-- Trigger: prevent loan if member already has 3 active loans
CREATE FUNCTION check_loan_limit() RETURNS TRIGGER AS $$
DECLARE
    active_count INTEGER;
BEGIN
    SELECT COUNT(*) INTO active_count
    FROM loans
    WHERE member_id = NEW.member_id
      AND returned_at IS NULL;

    IF active_count >= 3 THEN
        RAISE EXCEPTION 'Member % has reached the 3-loan limit', NEW.member_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_loan_limit
BEFORE INSERT ON loans
FOR EACH ROW EXECUTE FUNCTION check_loan_limit();

-- Trigger: audit log for order status changes
CREATE TABLE order_audit (
    id         SERIAL PRIMARY KEY,
    order_id   INTEGER,
    old_status TEXT,
    new_status TEXT,
    changed_at TIMESTAMP DEFAULT NOW()
);

CREATE FUNCTION log_order_status() RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status IS DISTINCT FROM NEW.status THEN
        INSERT INTO order_audit (order_id, old_status, new_status)
        VALUES (NEW.id, OLD.status, NEW.status);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_order_audit
AFTER UPDATE ON orders
FOR EACH ROW EXECUTE FUNCTION log_order_status();
----

=== Practice: Time-Series Event Log

[source,sql]
----
-- Parent table
CREATE TABLE events (
    id         BIGSERIAL,
    event_type TEXT NOT NULL,
    payload    JSONB,
    created_at TIMESTAMP NOT NULL DEFAULT NOW()
) PARTITION BY RANGE (created_at);

-- Monthly partitions
CREATE TABLE events_2025_01 PARTITION OF events
    FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');

CREATE TABLE events_2025_02 PARTITION OF events
    FOR VALUES FROM ('2025-02-01') TO ('2025-03-01');

CREATE TABLE events_2025_03 PARTITION OF events
    FOR VALUES FROM ('2025-03-01') TO ('2025-04-01');

-- Default partition catches anything outside defined ranges
CREATE TABLE events_default PARTITION OF events DEFAULT;

-- PostgreSQL will route inserts to the correct partition automatically
INSERT INTO events (event_type, payload, created_at)
VALUES ('page_view', '{"url":"/home"}', '2025-01-15 10:00:00');

-- Query pruning: Postgres only scans the relevant partition
EXPLAIN SELECT * FROM events WHERE created_at = '2025-01-20';

-- Detach old partition for archival (fast, no data copy)
ALTER TABLE events DETACH PARTITION events_2025_01;
----

== Phase 10 — Replication and High Availability (Week 10)

=== Goals
- Understand how PostgreSQL replication works
- Set up streaming replication with Docker

=== Concepts
- WAL (Write-Ahead Log) — the foundation of replication
- Streaming replication: primary → standby
- Synchronous vs asynchronous replication
- `pg_basebackup` for replica seeding
- `recovery.conf` / `postgresql.conf` settings
- Failover and `pg_promote()`
- Logical replication: replicate specific tables, useful for migrations

=== Docker Compose: Primary + Replica

[source,yaml]
----
services:
  pg_primary:
    image: postgres:17
    environment:
      POSTGRES_USER: replicator
      POSTGRES_PASSWORD: secret
      POSTGRES_DB: mydb
    command: >
      postgres
        -c wal_level=replica
        -c max_wal_senders=5
        -c wal_keep_size=64
    volumes:
      - primary_data:/var/lib/postgresql/data
    ports:
      - "5433:5432"

  pg_replica:
    image: postgres:17
    environment:
      PGUSER: replicator
      PGPASSWORD: secret
    command: >
      bash -c "
        until pg_basebackup -h pg_primary -D /var/lib/postgresql/data
          -U replicator -vP -Xs -R; do sleep 2; done &&
        postgres"
    volumes:
      - replica_data:/var/lib/postgresql/data
    depends_on:
      - pg_primary
    ports:
      - "5434:5432"

volumes:
  primary_data:
  replica_data:
----

== Phase 11 — Security and Access Control (Week 11)

=== Goals
- Follow least-privilege principles
- Use row-level security for multi-tenant data isolation

=== Concepts
- Roles and users: `CREATE ROLE`, `GRANT`, `REVOKE`
- Schema-level permissions
- `pg_hba.conf` — host-based authentication
- Row-Level Security (RLS): `ALTER TABLE ... ENABLE ROW LEVEL SECURITY`, `CREATE POLICY`
- `SECURITY DEFINER` vs `SECURITY INVOKER` functions
- Encrypting sensitive columns with `pgcrypto`

=== Practice

[source,sql]
----
-- Read-only reporting role
CREATE ROLE reporting_ro;
GRANT CONNECT ON DATABASE mydb TO reporting_ro;
GRANT USAGE ON SCHEMA public TO reporting_ro;
GRANT SELECT ON ALL TABLES IN SCHEMA public TO reporting_ro;
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT SELECT ON TABLES TO reporting_ro;

CREATE USER analyst WITH PASSWORD 'analyst_pass';
GRANT reporting_ro TO analyst;

-- Row-Level Security: multi-tenant table where users only see their rows
CREATE TABLE tenant_data (
    id        SERIAL PRIMARY KEY,
    tenant_id INTEGER NOT NULL,
    data      TEXT
);

ALTER TABLE tenant_data ENABLE ROW LEVEL SECURITY;

CREATE POLICY tenant_isolation ON tenant_data
    USING (tenant_id = current_setting('app.tenant_id')::INTEGER);

-- Application sets tenant context before queries
SET app.tenant_id = '42';
SELECT * FROM tenant_data; -- only sees tenant 42 rows

-- Encrypt sensitive data with pgcrypto
CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE user_secrets (
    id             SERIAL PRIMARY KEY,
    user_id        INTEGER,
    encrypted_ssn  BYTEA
);

INSERT INTO user_secrets (user_id, encrypted_ssn)
VALUES (1, pgp_sym_encrypt('123-45-6789', 'encryption_key'));

SELECT pgp_sym_decrypt(encrypted_ssn, 'encryption_key') AS ssn
FROM user_secrets WHERE user_id = 1;
----

== Reference: Useful System Queries

[source,sql]
----
-- Currently running queries
SELECT pid, now() - pg_stat_activity.query_start AS duration, query, state
FROM pg_stat_activity
WHERE state != 'idle'
ORDER BY duration DESC;

-- Kill a long-running query
SELECT pg_cancel_backend(<pid>);
SELECT pg_terminate_backend(<pid>); -- forceful

-- Database size
SELECT pg_size_pretty(pg_database_size('mydb'));

-- Table sizes with indexes
SELECT relname,
       pg_size_pretty(pg_total_relation_size(relid)) AS total_size,
       pg_size_pretty(pg_relation_size(relid))        AS table_size,
       pg_size_pretty(pg_indexes_size(relid))         AS index_size
FROM pg_catalog.pg_statio_user_tables
ORDER BY pg_total_relation_size(relid) DESC;

-- Unused indexes (candidates for removal)
SELECT relname, indexrelname, idx_scan
FROM pg_stat_user_indexes
WHERE idx_scan = 0
  AND indexrelname NOT LIKE 'pg_%';

-- Bloated tables needing VACUUM
SELECT relname, n_dead_tup, n_live_tup,
       round(n_dead_tup::NUMERIC / NULLIF(n_live_tup,0) * 100, 2) AS dead_ratio
FROM pg_stat_user_tables
ORDER BY dead_ratio DESC NULLS LAST;

-- Cache hit ratio (should be > 99% in production)
SELECT sum(heap_blks_read)  AS disk_reads,
       sum(heap_blks_hit)   AS cache_hits,
       round(sum(heap_blks_hit)::NUMERIC /
             NULLIF(sum(heap_blks_hit) + sum(heap_blks_read), 0) * 100, 2)
         AS cache_hit_ratio
FROM pg_statio_user_tables;
